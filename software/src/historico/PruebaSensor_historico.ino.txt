/*
 * ============================================================
 *  Detector de fugas para riego con dos sensores de flujo
 *  TEI201 - Taller de Diseño en Ingeniería · Prototipo Alpha
 * ============================================================
 *  Placa:     ESP32-S3 Super Mini (Arduino IDE 1.8.19,
 *             "ESP32S3 Dev Module", USB CDC On Boot: Enabled)
 *  Sensores:  2 x FS400A (1", 1-60 L/min), salida de pulsos
 *  Actuador:  3 LED (verde = normal, amarillo = posible fuga,
 *             rojo = fuga)
 *  Librerías: ninguna externa (solo el núcleo Arduino-ESP32)
 *
 *  Lógica:
 *   1. Cada sensor genera un pulso por cada fracción de litro.
 *   2. Se cuentan los pulsos de ambos sensores durante una
 *      ventana de 10 s y se convierten a caudal (L/min).
 *   3. Si el sensor 2 (aguas abajo) mide menos que el sensor 1
 *      (aguas arriba), el agua que falta se está perdiendo
 *      entre ambos: hay fuga.
 *   4. Según el % de diferencia se enciende el LED que
 *      corresponde.
 *
 *  Conexiones: GPIO1 = sensor 1, GPIO2 = sensor 2 (cada uno a
 *  través de un divisor 10k/20k que baja 5 V a ~3,3 V);
 *  GPIO4/5/6 = LED verde/amarillo/rojo con 330 ohm.
 * ============================================================
 */

// ---------- Pines ----------
const int PIN_S[2] = {1, 2};        // GPIO1 = sensor 1 (entrada), GPIO2 = sensor 2 (salida)
const int LED_VERDE = 4;
const int LED_AMARILLO = 5;
const int LED_ROJO = 6;

// ---------- Calibración ----------
// K = pulsos por litro. K1 es el dato del fabricante (f = 4,8·Q -> 288 pulsos/L).
// K2 se calibró comparando ambos sensores con la llave de fuga cerrada:
// razón de pulsos T2/T1 = 0,955 -> K2 = 288 x 0,955 ≈ 275.
float K[2] = {288.0, 275.0};

// ---------- Filtro anti-ruido ----------
// El motor de la bomba generaba picos de microsegundos que se contaban como pulsos.
// Se lee cada pin cada 0,2 ms y un cambio de nivel solo se acepta si se mantiene
// 5 lecturas seguidas (1 ms). Los pulsos reales del sensor duran decenas de ms.
const uint32_t MUESTRA_US = 200;    // período de muestreo [µs]
const int ESTABLE = 5;              // lecturas iguales necesarias para aceptar un cambio

// ---------- Decisión ----------
const unsigned long VENTANA_MS = 10000; // ventana de medición [ms]; más larga = más estable
const float Q_MIN = 0.5;                // [L/min] bajo esto se considera que no hay flujo
const float UMBRAL_AMARILLO = 10.0;     // [%] diferencia para "posible fuga"
const float UMBRAL_ROJO = 25.0;         // [%] diferencia para "fuga"

// ---------- Estado ----------
int nivel[2];                       // último nivel estable de cada sensor (HIGH/LOW)
int cuenta[2] = {0, 0};             // lecturas consecutivas distintas al nivel estable
uint32_t pulsos[2] = {0, 0};        // total acumulado de pulsos desde el arranque
uint32_t anterior[2] = {0, 0};      // total en la ventana anterior
uint32_t ultimaMuestra = 0;         // instante de la última lectura [µs]
unsigned long ultimoReporte = 0;    // instante del último cálculo [ms]

// Lee el sensor i y cuenta un pulso por cada flanco de subida estable.
void muestrear(int i) {
  int lectura = digitalRead(PIN_S[i]);
  if (lectura != nivel[i]) {
    cuenta[i]++;
    if (cuenta[i] >= ESTABLE) {     // el cambio duró al menos 1 ms: es real
      nivel[i] = lectura;
      cuenta[i] = 0;
      if (lectura == HIGH) {
        pulsos[i]++;
      }
    }
  } else {
    cuenta[i] = 0;                  // fue un pico de ruido: se descarta
  }
}

// Enciende solo los LED indicados.
void luces(bool v, bool a, bool r) {
  digitalWrite(LED_VERDE, v ? HIGH : LOW);
  digitalWrite(LED_AMARILLO, a ? HIGH : LOW);
  digitalWrite(LED_ROJO, r ? HIGH : LOW);
}

void setup() {
  Serial.begin(115200);

  pinMode(LED_VERDE, OUTPUT);
  pinMode(LED_AMARILLO, OUTPUT);
  pinMode(LED_ROJO, OUTPUT);
  luces(false, false, false);

  for (int i = 0; i < 2; i++) {
    pinMode(PIN_S[i], INPUT);       // el divisor externo fija el nivel; no se usa pull-up
    nivel[i] = digitalRead(PIN_S[i]);
  }

  ultimaMuestra = micros();
  ultimoReporte = millis();
}

void loop() {
  // 1) Muestreo de ambos sensores (no usar delay() en este programa)
  uint32_t ahora = micros();
  if (ahora - ultimaMuestra >= MUESTRA_US) {
    ultimaMuestra = ahora;
    muestrear(0);
    muestrear(1);
  }

  // 2) Cada ventana: calcular caudales, diferencia y estado
  if (millis() - ultimoReporte >= VENTANA_MS) {
    ultimoReporte = millis();

    uint32_t n1 = pulsos[0] - anterior[0];   // pulsos de la ventana
    uint32_t n2 = pulsos[1] - anterior[1];
    anterior[0] = pulsos[0];
    anterior[1] = pulsos[1];

    // Q [L/min] = pulsos por segundo x 60 / K
    float seg = VENTANA_MS / 1000.0;
    float q1 = n1 / seg * 60.0 / K[0];
    float q2 = n2 / seg * 60.0 / K[1];

    // Diferencia relativa: porcentaje del agua de entrada que no llega a la salida
    float dif = 0;
    if (q1 >= Q_MIN) {
      dif = (q1 - q2) / q1 * 100.0;
    }

    // 3) Actuador
    if (q1 < Q_MIN) {
      luces(false, false, false);          // sin flujo
    } else if (dif < UMBRAL_AMARILLO) {
      luces(true, false, false);           // normal
    } else if (dif < UMBRAL_ROJO) {
      luces(false, true, false);           // posible fuga
    } else {
      luces(false, false, true);           // fuga
    }

    // 4) Reporte por Monitor Serie (115200 baudios)
    Serial.print("Q1: ");
    Serial.print(q1, 2);
    Serial.print(" | Q2: ");
    Serial.print(q2, 2);
    Serial.print(" L/min | Dif: ");
    Serial.print(dif, 1);
    Serial.print(" % | Tot1: ");
    Serial.print(pulsos[0]);
    Serial.print(" | Tot2: ");
    Serial.print(pulsos[1]);
    Serial.print(" | Razon T2/T1: ");
    if (pulsos[0] > 0) {
      Serial.println((float)pulsos[1] / pulsos[0], 3);
    } else {
      Serial.println("-");
    }
  }
}
